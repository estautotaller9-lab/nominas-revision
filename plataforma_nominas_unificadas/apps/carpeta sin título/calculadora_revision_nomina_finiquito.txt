let workerBlobUrl=null;

async function obtenerWorkerSrc(){
  // Fix Safari: algunos Safari fallan cargando el worker de pdf.js directamente desde CDN
  // (problemas de CORS/tipo de worker). Lo descargamos y servimos como blob local.
  if(workerBlobUrl)return workerBlobUrl;
  try{
    const resp=await fetch('https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js');
    if(!resp.ok)throw new Error('HTTP '+resp.status);
    const code=await resp.text();
    workerBlobUrl=URL.createObjectURL(new Blob([code],{type:'application/javascript'}));
    return workerBlobUrl;
  }catch(e){
    // Si falla la descarga (p.ej. sin red), usamos la URL directa como último recurso
    return 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
  }
}

async function abrirDocumentoPDF(datos){
  let ultimoError=null;

  // Primer intento: worker convertido en blob. Es el que mejor funciona en Safari
  // cuando el HTML se abre directamente desde Descargas o Archivos.
  try{
    pdfjsLib.GlobalWorkerOptions.workerSrc=await obtenerWorkerSrc();
    return await pdfjsLib.getDocument({data:datos}).promise;
  }catch(e){
    ultimoError=e;
    console.warn('Primer intento PDF.js falló',e);
  }

  // Segundo intento: worker directo de CDN.
  try{
    pdfjsLib.GlobalWorkerOptions.workerSrc='https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
    return await pdfjsLib.getDocument({data:datos}).promise;
  }catch(e){
    ultimoError=e;
    console.warn('Segundo intento PDF.js falló',e);
  }

  // Último intento: modo sin worker. Es más lento, pero evita bloqueos de Safari.
  try{
    return await pdfjsLib.getDocument({data:datos,disableWorker:true}).promise;
  }catch(e){
    ultimoError=e;
  }
  throw ultimoError||new Error('No se pudo abrir el PDF');
}

async function leerPDF(ev){
  const file=ev.target.files&&ev.target.files[0];
  if(!file)return;
  setStatus('Leyendo PDF…');
  try{
    if(!window.pdfjsLib)throw new Error('No se pudo cargar el lector PDF. Comprueba la conexión a Internet.');
    const datos=await file.arrayBuffer();
    const pdf=await abrirDocumentoPDF(datos);
    let texto='';
    filasPDF=[];

    for(let p=1;p<=pdf.numPages;p++){
      const page=await pdf.getPage(p);
      const content=await page.getTextContent();
      const items=content.items
        .map(i=>({txt:String(i.str||''),x:Number(i.transform&&i.transform[4]||0),y:Number(i.transform&&i.transform[5]||0)}))
        .filter(i=>i.txt.trim()!=='');

      // Agrupa por altura con tolerancia. Algunas versiones de Safari devuelven
      // pequeñas diferencias decimales en la coordenada Y.
      const grupos=[];
      items.sort((a,b)=>b.y-a.y||a.x-b.x);
      for(const item of items){
        let grupo=grupos.find(g=>Math.abs(g.y-item.y)<=2.2);
        if(!grupo){
          grupo={y:item.y,items:[]};
          grupos.push(grupo);
        }
        grupo.items.push(item);
      }
      grupos.sort((a,b)=>b.y-a.y);

      for(const grupo of grupos){
        const row=grupo.items.sort((a,b)=>a.x-b.x);
        filasPDF.push(row);
        texto+=row.map(i=>i.txt).join(' ').replace(/\s+/g,' ').trim()+'\n';
      }
    }

    $('texto').value=texto.trim();
    analizarTexto();
  }catch(e){
    console.error(e);
    setStatus('No se pudo leer el PDF: '+(e&&e.message?e.message:e),'bad');
  }finally{
    ev.target.value='';
  }
}

function normalizarCabecera(s){
  return String(s||'')
    .toUpperCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g,'')
    .replace(/[^A-Z]/g,'');
}

function buscarItemCabecera(row,patrones){
  for(const it of row){
    const n=normalizarCabecera(it.txt);
    if(patrones.some(re=>re.test(n)))return it;
  }
  return null;
}

    const cDev=buscarItemCabecera(row,[/^DEVENGO$/,/^DEVENGOS$/]);
    const cDed=buscarItemCabecera(row,[/^DEDUCC$/,/^DEDUCCION$/,/^DEDUCCIONES$/]);

    if(cConc&&cUds&&cImp&&cDev&&cDed){

      return {

        codigo:cCod?cCod.x:null,

        concepto:cConc.x,
        devengo:cDev.x,
        deduccion:cDed.x,

  }
  return null;

        devengo:euros(textoColumna(row,bConceptoDev,bDevDed)),
        deduccion:euros(textoColumna(row,bDevDed,99999))
      };

      // En algunas nóminas el PRECIO está vacío (carrera profesional, pagas, etc.).
      // Si el concepto ha quedado vacío o incompleto, reconstruimos la fila por texto.
      if(!fila.concepto || fila.concepto.length<4){
        const textoFila=row.map(i=>i.txt).join(' ').replace(/\s+/g,' ').trim();
        const mm=textoFila.match(/^(\d+(?:[,.]\d+)?)?\s*(?:\d+(?:\.\d{3})*(?:,\d+)?)?\s*(.+?)\s+(\d{1,3}(?:\.\d{3})*,\d{2})$/);
        if(mm){
          fila.uds=mm[1]?euros(mm[1]):fila.uds;
          fila.concepto=mm[2].trim();
          fila.devengo=euros(mm[3]);
        }
      }
    }else{
      const codigoX=cols.codigo===null?0:cols.codigo;
      const bCodConc=cols.codigo===null?cols.concepto-40:(cols.codigo+cols.concepto)/2;
      const bConcUds=(cols.concepto+cols.unidades)/2;
      const bUdsPrecio=(cols.unidades+cols.precio)/2;
      const bPrecioDev=(cols.precio+cols.devengo)/2;
      const bDevDed=(cols.devengo+cols.deduccion)/2;

      fila={
        codigo:cols.codigo===null?'':textoColumna(row,Math.max(0,codigoX-30),bCodConc),
        concepto:textoColumna(row,bCodConc,bConcUds),

        importe:euros(textoColumna(row,bUdsPrecio,bPrecioDev)),
        devengo:euros(textoColumna(row,bPrecioDev,bDevDed)),
        deduccion:euros(textoColumna(row,bDevDed,99999))
      };

    fila.concepto=String(fila.concepto||'').replace(/\s+/g,' ').trim();

  }
  return filas;
}

    {re:/FESTIVO TRABAJADO/i,tipo:'devengo'},
    {re:/PAGA EXTRA (?:VERANO|NAVIDAD|BENEFICIOS)/i,tipo:'devengo'},
    {re:/PRESTACI[ÓO]N (?:SS|ENFERMEDAD|IT)(?:\s*75%|\s*60%)?/i,tipo:'devengo'},
    {re:/COMPLEMENTO (?:IT|ENFERMEDAD)|REGULARIZACI[ÓO]N PROCESO IT/i,tipo:'devengo'},
    {re:/VACACIONES LIQUIDADAS/i,tipo:'devengo'},
    {re:/NOCTURNIDAD/i,tipo:'devengo'},
    {re:/PELIGROSIDAD/i,tipo:'devengo'},
    {re:/TRAB\.?\s*CONT\.?\s*COMUNES|COTIZACION REGIMEN GENERAL/i,tipo:'deduccion'},
    {re:/TRAB\.?\s*DESEMPLEO|COTIZACION DESEMPLEO/i,tipo:'deduccion'},
    {re:/TRAB\.?\s*FORMAC\.?\s*PROFESIONAL|FORMACION PROFESIONAL/i,tipo:'deduccion'},
    {re:/\bMEI\b/i,tipo:'deduccion'},
    {re:/RETENCI[ÓO]N A CTA\.?\s*IRPF|RET\.?\s*A CUENTA DEL I\.?R\.?P\.?F\.?/i,tipo:'deduccion'}
  ];

  const resultado=[];
  for(const linea of lineas){
    const patron=patrones.find(p=>p.re.test(linea));
    if(!patron)continue;

    const matchConcepto=linea.match(patron.re);
    if(!matchConcepto)continue;
    const inicio=matchConcepto.index;
    const fin=inicio+matchConcepto[0].length;
    const antes=linea.slice(0,inicio);
    const despues=linea.slice(fin);

    const numerosAntes=[...antes.matchAll(/\d{1,3}(?:\.\d{3})*,\d{2}|\d+(?:,\d{2})/g)].map(x=>euros(x[0]));
    const numerosDespues=[...despues.matchAll(/\d{1,3}(?:\.\d{3})*,\d{2}|\d+(?:,\d{2})/g)].map(x=>euros(x[0]));
    if(!numerosDespues.length)continue;

    const importeFinal=numerosDespues[numerosDespues.length-1];
    const fila={
      codigo:'',
      concepto:matchConcepto[0].replace(/\s+/g,' ').trim().toUpperCase(),
      uds:numerosAntes[0]||0,
      importe:numerosAntes[1]||0,
      devengo:patron.tipo==='devengo'?importeFinal:0,
      deduccion:patron.tipo==='deduccion'?importeFinal:0
    };

    // Formato antiguo con código delante: 010 SALARIO BASE 1 1.352,07 1.352,07
    const cod=linea.match(/^\s*(\d{3})\b/);
    if(cod){
      fila.codigo=cod[1];
      const nums=[...linea.slice(fin).matchAll(/\d{1,3}(?:\.\d{3})*,\d{2}|\d+(?:,\d{2})/g)].map(x=>euros(x[0]));
      if(nums.length>=3){
        fila.uds=nums[0];
        fila.importe=nums[1];
        if(patron.tipo==='devengo')fila.devengo=nums[nums.length-1];
        else fila.deduccion=nums[nums.length-1];
      }else if(nums.length===2){
        fila.importe=nums[0];
        if(patron.tipo==='devengo')fila.devengo=nums[1];
        else fila.deduccion=nums[1];
      }
    }

    resultado.push(fila);
  }
  return resultado;
}

function fusionarFilasConcepto(coordenadas,texto){
  const claves=f=>String(f.concepto||'')
    .toUpperCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g,'')
    .replace(/\bPLUS\b/g,'')
    .replace(/[^A-Z]/g,'');

  const mapa=new Map();
  for(const f of [...coordenadas,...texto]){
    const k=claves(f);
    if(!k)continue;
    const actual=mapa.get(k);
    if(!actual){
      mapa.set(k,{...f});
      continue;
    }
    // Conserva el dato más completo de cada columna.
    for(const campo of ['codigo','concepto','uds','importe','devengo','deduccion']){
      if((actual[campo]===0 || actual[campo]==='' || actual[campo]==null) && f[campo]){
        actual[campo]=f[campo];
      }
    }
    // El parser por texto es especialmente fiable para el importe final.
    if(f.devengo)actual.devengo=f.devengo;
    if(f.deduccion)actual.deduccion=f.deduccion;
  }
  return [...mapa.values()];
}

function pintarTablaConceptos(filas){
  if(!filas.length){$('tablaConceptos').innerHTML='<tr><td colspan="7" class="small">No se ha detectado la tabla de conceptos en este PDF. Revisa el texto extraído o completa los importes a mano en "Nómina leída".</td></tr>';return}
  $('tablaConceptos').innerHTML=filas.map(f=>{
    let esperado=null;
    let importeReal=f.devengo||f.deduccion||0;
    if(f.uds&&f.importe){
      esperado=f.deduccion ? (f.uds*f.importe/100) : (f.uds*f.importe);
    }
    let badge='<span class="badge b-warn">Sin unidades/precio</span>';
    if(esperado!==null){
      const ok=Math.abs(esperado-importeReal)<=0.06;
      badge=`<span class="badge ${ok?'b-ok':'b-bad'}">${ok?'Cuadra':'Revisar ('+money(esperado)+')'}</span>`;
    }
    return `<tr><td>${esc(f.codigo)}</td><td>${esc(f.concepto)}</td><td class="num">${f.uds||''}</td><td class="num">${f.importe?money(f.importe):''}</td><td class="num">${f.devengo?money(f.devengo):''}</td><td class="num">${f.deduccion?money(f.deduccion):''}</td><td>${badge}</td></tr>`;
  }).join('');
}

function mapearConceptos(filas){
  const suma=(re,campo)=>filas
    .filter(f=>re.test(String(f.concepto||'')))
    .reduce((s,f)=>s+Number(f[campo]||0),0);

  const salario=suma(/SALARIO BASE/i,'devengo');
  const antig=suma(/ANTIG[ÜU]EDAD/i,'devengo');
  const carrera=suma(/CARRERA PROFESIONAL/i,'devengo');
  const noct=suma(/NOCTURNID/i,'devengo');
  const pelig=suma(/PELIGROSID/i,'devengo');
  const domingos=suma(/DOMINGO TRABAJADO/i,'devengo');
  const festivos=suma(/COMPLEMENTO FESTIVO|FESTIVO TRABAJADO/i,'devengo');
  const pagaExtra=suma(/PAGA EXTRA/i,'devengo');
  const prestacionIT=suma(/PRESTACI[ÓO]N (?:SS|ENFERMEDAD|IT)/i,'devengo');
  const complementoIT=suma(/COMPLEMENTO (?:IT|ENFERMEDAD)|REGULARIZACI[ÓO]N PROCESO IT/i,'devengo');
  const irpf=suma(/IRPF/i,'deduccion');
  const cc=suma(/CONT\.?\s*COMUNES|REGIMEN GENERAL/i,'deduccion');
  const desempleo=suma(/DESEMPLEO/i,'deduccion');
  const fp=suma(/FORMAC\.?\s*PROFESIONAL|FORMACION PROFESIONAL/i,'deduccion');
  const mei=suma(/\bMEI\b/i,'deduccion');

  set('salarioPagado',salario);
  set('antiguedadPagada',antig);
  set('carreraPagada',carrera);
  set('nocturnidadPagada',noct);
  set('peligrosidadPagada',pelig);
  set('domFestPagado',domingos+festivos);
  set('pagaExtraLeida',pagaExtra);
  set('prestacionITLeida',prestacionIT);
  set('complementoITLeido',complementoIT);

  // Rellena automáticamente el número de domingos/festivos trabajados para que
  // el "Cuadre final del mes" no marque una diferencia falsa por no haberlos
  // pasado a mano. El plus por domingo es un importe fijo (coincide con lo leído
  // en la nómina), así que la división es exacta. El de festivo, en cambio, se paga
  // como (salario/día × 1,75) + complemento, así que solo se rellena si la
  // división por ese importe estimado da un número de festivos razonablemente
  // entero; si no cuadra bien se deja en blanco para que se revise a mano.
  if($('plusDomingo') && domingos>0){
    const plusDomingoRate=n('plusDomingo')||49.96;
    const domingosCount=plusDomingoRate>0?Math.round(domingos/plusDomingoRate):0;
    if(domingosCount>0 && Math.abs(domingosCount*plusDomingoRate-domingos)<=0.05){
      set('domingosMes',domingosCount);
    }
  }
  if($('plusFestivo') && festivos>0){
    const salarioDiaEstimado=(salario+antig+carrera+noct+pelig)/30;
    const festivoUnitarioEstimado=(salarioDiaEstimado*1.75)+(n('plusFestivo')||49.96);
    const festivosCount=festivoUnitarioEstimado>0?Math.round(festivos/festivoUnitarioEstimado):0;
    if(festivosCount>0 && Math.abs(festivosCount*festivoUnitarioEstimado-festivos)<=1){
      set('festivosMes',festivosCount);
    }
  }
  set('totalITLeido',prestacionIT+complementoIT);
  set('irpfRetenido',irpf);
  set('ccRetenido',cc);
  set('desempleoRetenido',desempleo);
  set('fpRetenido',fp);
  set('meiRetenido',mei);

  const excluir=/SALARIO BASE|ANTIG[ÜU]EDAD|CARRERA PROFESIONAL|NOCTURNID|PELIGROSID|DOMINGO|FESTIVO|PAGA EXTRA|BENEFICIOS|VACACION|PRESTACI[ÓO]N (?:SS|ENFERMEDAD|IT)|COMPLEMENTO (?:IT|ENFERMEDAD)|REGULARIZACI[ÓO]N PROCESO IT|COTIZABLES|NO COTIZABLES|IRPF|CONT\.?\s*COMUNES|REGIMEN GENERAL|DESEMPLEO|FORMAC|MEI|TOTAL DEVENGOS|COSTE SEG|INFORMATIVO/i;
  const otros=filas
    .filter(f=>!excluir.test(String(f.concepto||'')))
    .reduce((s,f)=>s+Number(f.devengo||0),0);
  set('otrosDevengos',otros);

  const filaIrpf=filas.find(f=>/IRPF/i.test(String(f.concepto||'')));
  if(filaIrpf&&filaIrpf.uds)set('irpfPct',filaIrpf.uds);

  const ssPct=filas
    .filter(f=>/CONT\.?\s*COMUNES|REGIMEN GENERAL|DESEMPLEO|FORMAC\.?\s*PROFESIONAL|\bMEI\b/i.test(String(f.concepto||'')))
    .reduce((s,f)=>s+Number(f.uds||0),0);
  if(ssPct)set('ssPct',ssPct);

  const totalRow=filas.find(f=>/TOTAL DEVENGOS?\s*Y\s*DEDUCCI/i.test(String(f.concepto||'')));
  if(totalRow){
    set('totalDevNomina',totalRow.devengo);
    set('totalDedNomina',totalRow.deduccion);
  }

  const vacRow=filas.find(f=>/VACACION.*LIQUID/i.test(String(f.concepto||'')));
  if(vacRow)set('vacPagadas',vacRow.uds||0);
}

function analizarTexto(){
  const t=$('texto').value.replace(/\u00a0/g,' ');

    else if(/ENCARGADO/.test(c))set('categoria','ENCARGADO DE GRUPO');
    else if(/ESPECIALISTA/.test(c))set('categoria','ESPECIALISTA');

  // --- Bloque empresa ---
  m=flat.match(/\bEmpresa\s+(.+?)(?=\s+C\.?\s*I\.?\s*F\.?|\s+Centro)/i)||flat.match(/\bEMPRESA\s*:?\s*(.+?)(?=\s+DOMICILIO)/i);if(m)set('empresa',m[1].trim());

  // La fila numérica contiene normalmente siete importes.
  const filaTotales=ls.find(x=>{
    const a=[...x.matchAll(/\d{1,3}(?:\.\d{3})*,\d{2}/g)];
    return a.length>=7;
  });
  if(filaTotales){
    const a=[...filaTotales.matchAll(/\d{1,3}(?:\.\d{3})*,\d{2}/g)].map(x=>euros(x[0]));
    set('baseIrpf',a[2]||0);
    set('totalDevNomina',a[a.length-2]||0);
    set('totalDedNomina',a[a.length-1]||0);
  }

  // IRPF y Seguridad Social reales leídos de las filas de deducción.
  if(filasConcepto.length){
    const filaIrpf=filasConcepto.find(f=>/IRPF/i.test(f.concepto));
    if(filaIrpf&&filaIrpf.uds)set('irpfPct',filaIrpf.uds);
    const ssLeida=filasConcepto
      .filter(f=>/CONT\.?\s*COMUNES|DESEMPLEO|FORMAC\.?\s*PROFESIONAL|\bMEI\b/i.test(f.concepto))
      .reduce((s,f)=>s+Number(f.uds||0),0);
    if(ssLeida)set('ssPct',ssLeida);
  }

  // Líquido a percibir (footer: TRANSFERIDO A / BASE IRPF / % IRPF / LÍQUIDO TOTAL A PERCIBIR)
  m=flat.match(/L[ÍI]QUIDO\s*TOTAL(?:\s*A\s*PERCIBIR)?\s+(\d{1,3}(?:\.\d{3})*,\d{2})/i);
  if(m)set('liquidoNomina',euros(m[1]));
  if(!n('liquidoNomina')&&n('totalDevNomina'))set('liquidoNomina',n('totalDevNomina')-n('totalDedNomina'));

  const nivelDetectado=detectarNivelCarrera(n('carreraPagada'));
  if(nivelDetectado)$('nivelCarrera').value=String(nivelDetectado);
  const espCarga=datosEsperadosNomina();
  const nivelTxt=espCarga.nivel?`Nivel ${nombreNivelCarrera(espCarga.nivel)}`:'sin nivel';
  const salarioOk=Math.abs(n('salarioPagado')-espCarga.salarioPeriodo)<=0.10;
  const antigOk=Math.abs(n('antiguedadPagada')-espCarga.antigPeriodo)<=0.15;
  setStatus(`Datos cargados: salario ${money(n('salarioPagado'))}, IT ${money(n('prestacionITLeida')+n('complementoITLeido'))}, domingos y festivos editables manualmente.`,'ok');
  sincronizarDatosFiniquito();
  abrirTab('conceptos',document.querySelectorAll('.tab')[0]);
  revisarTodo();
}


function detectarNivelCarrera(importe){
  const valor=Number(importe)||0;
  if(!valor)return 0;
  let mejor=0,distancia=Infinity;
  for(const [nivel,tabla] of Object.entries(CARRERA_2026)){
    const d=Math.abs(valor-tabla);
    if(d<distancia){distancia=d;mejor=Number(nivel)}
  }
  // Tolerancia amplia para pequeñas diferencias de redondeo como 306,95 como importe exacto.
  return distancia<=0.60?mejor:0;
}

function nombreNivelCarrera(nivel){
  return ['','I','II','III','IV'][Number(nivel)]||'';
}

// --- ÚNICA función que calcula salario, antigüedad, carrera profesional,
// domingos y festivos. La usan tanto la revisión de nómina como la
// comparación manual y el informe imprimible, para que una corrección aquí
// se aplique automáticamente en los tres sitios y no se quede coja en ninguno. ---
function calcularConceptosProtegidos(p){
  const y=p.finPeriodo.getFullYear();
  const tabla=TABLAS[y]||TABLAS[2026];
  const salarioMensual=(tabla[p.categoria]||0)*(p.jornada||1);
  const trienios=Math.max(0,Math.floor(yearsBetween(p.antiguedadFecha,p.finPeriodo)/3));
  const antigMensual=salarioMensual*((p.trienioPct||6)/100)*trienios;
  const carreraMensual=(p.aplicaCarrera&&y===2026&&p.nivelCarrera)?(CARRERA_2026[p.nivelCarrera]||0)*(p.jornada||1):0;
  const nocturnidadMensual=p.nocturnidadMensual||0;
  const peligrosidadMensual=p.peligrosidadMensual||0;

  const incluirSalario=p.incluirSalario!==false;
  const incluirAntiguedad=p.incluirAntiguedad!==false;
  const protegidoMensual=(incluirSalario?salarioMensual:0)+(incluirAntiguedad?antigMensual:0)+carreraMensual+nocturnidadMensual+peligrosidadMensual;
  const valorDia=protegidoMensual/30;

  const plusDomingoRate=p.plusDomingoRate||49.96;
  const plusFestivoRate=p.plusFestivoRate||49.96;
  const plusFestivoEspecialRate=p.plusFestivoEspecialRate||99.92;

  const domingoImporte=(p.domingosCount||0)*plusDomingoRate;
  const festivoUnit=(valorDia*1.75)+plusFestivoRate;
  const festivoEspecialUnit=(valorDia*1.75)+plusFestivoEspecialRate;
  const festivoImporte=(p.festivosCount||0)*festivoUnit;
  const festivoEspecialImporte=(p.festivosEspecialesCount||0)*festivoEspecialUnit;
  const totalDomFest=domingoImporte+festivoImporte+festivoEspecialImporte;

  return {
    y,tabla,salarioMensual,trienios,antigMensual,carreraMensual,
    nocturnidadMensual,peligrosidadMensual,protegidoMensual,valorDia,
    domingoImporte,festivoImporte,festivoEspecialImporte,totalDomFest,
    festivoUnit,festivoEspecialUnit
  };
}


function datosEsperadosNomina(){
  const finPeriodo=fecha($('periodoHasta').value)||fecha($('finContrato').value)||new Date();
  const jornada=1;
  const desde=fecha($('periodoDesde').value),hasta=fecha($('periodoHasta').value);
  const diasPeriodo=desde&&hasta?diasEntre(desde,hasta):30;

  const categoriaCalculo=($('categoriaFiniquito')&&$('categoriaFiniquito').value)||$('categoria').value;
  const antFecha=fecha($('antiguedadFecha').value);
  const pct=Number($('trienioPct').value)||6;

  let nivel=Number(($('nivelCarreraFiniquito')&&$('nivelCarreraFiniquito').value)||$('nivelCarrera').value||0);
  if(!nivel)nivel=detectarNivelCarrera(n('carreraPagada'));
  const aplicaCarrera=!$('aplicarCarrera') || $('aplicarCarrera').value==='si';

  const c=calcularConceptosProtegidos({
    finPeriodo,jornada,categoria:categoriaCalculo,antiguedadFecha:antFecha,
    trienioPct:pct,nivelCarrera:nivel,aplicaCarrera
  });

  return {
    y:c.y,jornada,diasPeriodo,salarioMensual:c.salarioMensual,salarioPeriodo:c.salarioMensual*(diasPeriodo/30),
    trienios:c.trienios,antigMensual:c.antigMensual,antigPeriodo:c.antigMensual*(diasPeriodo/30),
    nivel,carreraMensual:c.carreraMensual,carreraPeriodo:c.carreraMensual*(diasPeriodo/30)
  };
}

function lineaRevision(nombre,pagado,esperado,tolerancia=0.05){
  const dif=pagado-esperado;
  const ok=Math.abs(dif)<=tolerancia;
  return `<tr><td>${esc(nombre)}</td><td class="num">${money(pagado)}</td><td class="num">${money(esperado)}</td><td class="num">${money(dif)}</td><td><span class="badge ${ok?'b-ok':'b-bad'}">${ok?'Correcto':'Revisar'}</span></td></tr>`;
}

function diasUltimoMesCalculados(fin){
  if(!fin)return 0;
  if($('reglaUltimoMes').value==='naturales'){
    const inicioMes=new Date(fin.getFullYear(),fin.getMonth(),1);
    return diasEntre(inicioMes,fin);
  }
  // En nómina mensual se trabaja sobre meses de 30 días:
  // del día 1 al 23 = 23 días; los días 31 no superan 30.
  return Math.min(30,fin.getDate());
}

function proporcionPeriodo(inicioContrato,finContrato,desde,hasta){
  if(!inicioContrato||!finContrato)return 0;
  const ini=inicioContrato>desde?inicioContrato:desde;
  const fin=finContrato<hasta?finContrato:hasta;
  if(ini>fin)return 0;
  return diasEntre(ini,fin)/diasEntre(desde,hasta);
}

function calcularComparacionManual(){
  const periodo=fecha($('periodoHasta').value)||new Date();
  const categoria=$('manualCategoria').value;
  const ant=fecha($('manualAntiguedadFecha').value);
  const nivel=Number($('manualNivelCarrera').value||0);
  const noct=n('manualNocturnidad');
  const pelig=n('manualPeligrosidad');

  const c=calcularConceptosProtegidos({
    finPeriodo:periodo,jornada:1,categoria,antiguedadFecha:ant,
    trienioPct:Number($('manualTrienioPct').value||6),nivelCarrera:nivel,aplicaCarrera:true,
    nocturnidadMensual:noct,peligrosidadMensual:pelig,
    domingosCount:n('manualDomingos'),festivosCount:n('manualFestivos'),festivosEspecialesCount:n('manualFestivosEspeciales'),
    plusDomingoRate:n('plusDomingo'),plusFestivoRate:n('plusFestivo'),plusFestivoEspecialRate:n('plusFestivoEspecial')
  });
  const salario=c.salarioMensual, antig=c.antigMensual, carrera=c.carreraMensual, trienios=c.trienios;
  const domingo=c.domingoImporte, festivos=c.festivoImporte, festivosEsp=c.festivoEspecialImporte, totalDomFest=c.totalDomFest;

  const prestacionIT=n('prestacionITLeida');
  const complementoIT=n('complementoITLeido');
  const totalITPagado=prestacionIT+complementoIT;
  const esNominaIT=totalITPagado>0;

  let filas='';
  let esperado=0;
  let pagado=0;

  if(esNominaIT){
    // En una baja completa el salario, antigüedad, carrera, etc. no aparecen
    // como líneas separadas: quedan sustituidos por prestación + complemento.
    const protegido=c.protegidoMensual;
    esperado=protegido+totalDomFest;
    pagado=totalITPagado+n('domFestPagado');

    filas+=`<tr><td colspan="5"><strong>Nómina de baja detectada: se compara el total mensual protegido con Prestación IT + Complemento IT.</strong></td></tr>`;
    filas+=filaComparacionManual('Salario base protegido',salario,salario);
    filas+=filaComparacionManual(`Antigüedad protegida (${trienios} trienios)`,antig,antig,0.15);
    filas+=filaComparacionManual('Carrera protegida',carrera,carrera);
    filas+=filaComparacionManual('Nocturnidad protegida',noct,noct);
    filas+=filaComparacionManual('Peligrosidad protegida',pelig,pelig);
    filas+=`<tr><td><strong>Total mensual protegido</strong></td><td class="num"><strong>${money(protegido)}</strong></td><td class="num">—</td><td class="num">—</td><td><span class="badge b-ok">Referencia</span></td></tr>`;
    filas+=`<tr><td>Prestación IT</td><td class="num">—</td><td class="num">${money(prestacionIT)}</td><td class="num">—</td><td><span class="badge b-ok">Leída</span></td></tr>`;
    filas+=`<tr><td>Complemento de enfermedad</td><td class="num">—</td><td class="num">${money(complementoIT)}</td><td class="num">—</td><td><span class="badge b-ok">Leído</span></td></tr>`;
    filas+=filaComparacionManual('Cuadre de IT',protegido,totalITPagado,0.15);
    filas+=filaComparacionManual('Domingos',domingo,n('domFestPagado'));
    filas+=filaComparacionManual('Festivos ordinarios',festivos,0);
    filas+=filaComparacionManual('Festivos especiales',festivosEsp,0);
  }else{
    esperado=salario+antig+carrera+noct+pelig+totalDomFest;
    pagado=
      n('salarioPagado')+
      n('antiguedadPagada')+
      n('carreraPagada')+
      n('nocturnidadPagada')+
      n('peligrosidadPagada')+
      n('domFestPagado');

    filas+=filaComparacionManual('Salario base',salario,n('salarioPagado'));
    filas+=filaComparacionManual(`Antigüedad (${trienios} trienios)`,antig,n('antiguedadPagada'),0.15);
    filas+=filaComparacionManual('Carrera profesional',carrera,n('carreraPagada'));
    filas+=filaComparacionManual('Nocturnidad',noct,n('nocturnidadPagada'));
    filas+=filaComparacionManual('Peligrosidad',pelig,n('peligrosidadPagada'));
    filas+=filaComparacionManual('Domingos',domingo,n('domFestPagado'));
    filas+=filaComparacionManual('Festivos ordinarios',festivos,0);
    filas+=filaComparacionManual('Festivos especiales',festivosEsp,0);
  }

  $('tablaComparacionManual').innerHTML=filas;

  const dif=esperado-pagado;
  const irpf=esperado*n('manualIRPF')/100;
  const ss=esperado*0.065;
  const neto=esperado-irpf-ss;

  const estado=Math.abs(dif)<=0.15
    ? 'CUADRA'
    : dif>0
      ? 'FALTA POR PAGAR'
      : 'SE HA PAGADO DE MÁS';

  $('resumenComparacionManual').innerHTML=
    `<strong>Tipo de comprobación:</strong> ${esNominaIT?'Incapacidad temporal':'Nómina ordinaria'}.<br>
     <strong>Debería cobrar:</strong> ${money(esperado)}.<br>
     <strong>Le han pagado en conceptos comparables:</strong> ${money(pagado)}.<br>
     <strong>${estado}: ${money(Math.abs(dif))}</strong><br>
     <span class="small">Líquido estimado orientativo con SS 6,50 % e IRPF ${n('manualIRPF').toFixed(2)} %: ${money(neto)}.</span>`;
}

Si la persona SÍ tenía algún complemento, indícalo arriba (nivel de carrera, nocturnidad, peligrosidad) con una nómina normal suya de un mes trabajado y vuelve a calcular antes de concluir que se ha cobrado de más o de menos.</div></td></tr>`;

  $('mBruto').textContent=money(bruto);

  $('mNeto').textContent=money(neto);

    ['Duración del contrato',inicio&&fin?`${diasEntre(inicio,fin)} días`:'Fechas incompletas'],
    ['Último mes calculado',`${diasUltimoMes.toFixed(2)} días · ${money(salarioUltimoMes)}`],
    ['Domingos trabajados',`${n('domingosMes')} × ${money(n('plusDomingo'))} = ${money(domingosManual)}`],

    ['Salario y complementos pendientes',money(salarioPendiente)],
    ['Prestación IT leída',money(incluido('incPrestacionIT')?n('prestacionITLeida'):0)],
    ['Complemento IT leído',money(incluido('incComplementoIT')?n('complementoITLeido'):0)],
    ['Total IT abonado',money((incluido('incPrestacionIT')?n('prestacionITLeida'):0)+(incluido('incComplementoIT')?n('complementoITLeido'):0))],

  $('resumenNomina').innerHTML=[
    ['Categoría aplicada',($('categoriaFiniquito')&&$('categoriaFiniquito').options[$('categoriaFiniquito').selectedIndex].text)||$('categoria').options[$('categoria').selectedIndex].text],

    ['Carrera profesional aplicada',estadoCarrera],
    ['Nocturnidad aplicada',estadoNocturnidad],

// --- Integración con el hub "Nóminas Unificadas" ---
// Si esta calculadora se abre desde el hub (botón "Abrir" tras seleccionar
// un compañero/a), llega con parámetros en la URL: nombre, dni, turno,
// categoria, antiguedad, periodo. Los recogemos y precargamos lo que se pueda.
function mapCategoriaHub(txt){
  const t=(txt||'').toLowerCase();
  if(t.includes('encargado'))return 'ENCARGADO DE GRUPO';
  if(t.includes('especialista'))return 'ESPECIALISTA';
  if(t.includes('pe') && t.includes('especial'))return 'PEON ESPECIALIZADO';
  if(t.includes('limpiador'))return 'LIMPIADOR-A';
  return '';
}
function precargarDesdeHub(){
  const q=new URLSearchParams(window.location.search);
  if(![...q.keys()].length)return;
  if(q.get('nombre'))set('trabajador',q.get('nombre'));
  if(q.get('dni'))set('nif',q.get('dni'));
  const cat=mapCategoriaHub(q.get('categoria'));
  if(cat){set('categoria',cat);if($('manualCategoria'))set('manualCategoria',cat)}
  if(q.get('antiguedad')){
    set('antiguedadFecha',q.get('antiguedad'));
    if($('manualAntiguedadFecha'))set('manualAntiguedadFecha',q.get('antiguedad'));
  }
  const periodo=q.get('periodo');
  if(periodo){
    const meses={ENERO:1,FEBRERO:2,MARZO:3,ABRIL:4,MAYO:5,JUNIO:6,JULIO:7,AGOSTO:8,SEPTIEMBRE:9,OCTUBRE:10,NOVIEMBRE:11,DICIEMBRE:12};
    const m=periodo.match(/([A-Za-zñÑ]+)\s+(\d{4})/);
    if(m && meses[m[1].toUpperCase()]){
      const mes=meses[m[1].toUpperCase()],y=+m[2];
      const ultimoDia=new Date(y,mes,0).getDate();
      set('periodoDesde',iso(1,mes,y));
      set('periodoHasta',iso(ultimoDia,mes,y));
    }
  }
  setStatus('Datos del compañero/a precargados desde Nóminas Unificadas. Completa el resto y lee su nómina.','ok');
  if(typeof sincronizarDatosFiniquito==='function')sincronizarDatosFiniquito();
  revisarTodo();
}
precargarDesdeHub();
